## Custom layout {#composition}

Arrange namespace components inside the native form. The layout owns which inputs appear; see [rendering](../core-concepts#rendering).

```tsx
import { Form } from '@canard/schema-form';

const schema = {
  type: 'object',
  properties: {
    title: { type: 'string' },
    count: { type: 'number' },
    enabled: { type: 'boolean' },
  },
} as const;

export function ComposedForm() {
  return (
    <Form jsonSchema={schema}>
      {({ value }) => (
        <>
          <Form.Label path="/title" />
          <Form.Input path="/title" />
          <Form.Error path="/title" />
          <Form.Input path="/count" />
          <Form.Input path="/enabled" />
          <output>{JSON.stringify(value)}</output>
          <button type="submit">Submit</button>
        </>
      )}
    </Form>
  );
}
```

## Custom FieldInput {#custom-input}

Forward input drafts through onChange and honor resolved controls. Use a fieldInputMap entry instead of a definition when only one path needs an override; see [selection](../schema-reference#input-resolution).

```tsx
import { Form, type FieldInputProps } from '@canard/schema-form';

function TextInput({
  path, value, onChange, readOnly, disabled, placeholder, className,
}: FieldInputProps<string>) {
  return (
    <textarea
      id={path}
      value={value ?? ''}
      readOnly={readOnly}
      disabled={disabled}
      placeholder={placeholder}
      className={className}
      onChange={(event) => onChange(event.target.value || undefined)}
    />
  );
}

const schema = {
  type: 'object',
  properties: { text: { type: 'string', title: 'Text' } },
} as const;

export function CustomInputForm() {
  return (
    <Form
      jsonSchema={schema}
      fieldInputDefinitions={[{ test: { type: 'string' }, Component: TextInput }]}
    />
  );
}
```

## Submit with a typed handle {#submit}

The button calls the promise-returning handle and catches rejection. Validation issues stay on the form; other failures appear in the output. Explicit Value parameters keep the defaultValue and ref types aligned; see [inference](../api-reference#typescript).

```tsx
import { useRef, useState } from 'react';
import {
  Form, type FormHandle, isValidationError, useFormSubmit,
} from '@canard/schema-form';
import { plugin } from '@canard/schema-form-ajv8-plugin';

const schema = {
  type: 'object',
  properties: { text: { type: 'string', minLength: 3 } },
  required: ['text'],
} as const;
type Value = { text?: string };

export function SubmitForm() {
  const formRef = useRef<FormHandle<typeof schema, Value>>(null);
  const { submit, pending } = useFormSubmit(formRef);
  const [failure, setFailure] = useState('');

  async function save() {
    setFailure('');
    try {
      await submit();
    } catch (error) {
      if (!isValidationError(error)) setFailure(String(error));
    }
  }

  return (
    <>
      <Form<typeof schema, Value>
        ref={formRef}
        jsonSchema={schema}
        defaultValue={{ text: 'Ready' }}
        validatorFactory={plugin.validator}
        onSubmit={(value) => console.log(value)}
      />
      <button type="button" disabled={pending} onClick={save}>Save</button>
      <button type="button" onClick={() => formRef.current?.reset()}>Reset</button>
      <output>{failure}</output>
    </>
  );
}
```
